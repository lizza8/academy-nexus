#include <stdint.h>

typedef struct {
    uint8_t r, g, b;
} RGB;

RGB lerp(RGB a, RGB b, float t) {
    RGB out = {
        (uint8_t)(a.r + t * (b.r - a.r)),
        (uint8_t)(a.g + t * (b.g - a.g)),
        (uint8_t)(a.b + t * (b.b - a.b)),
    };
    return out;
}

const RGB = struct { r: u8, g: u8, b: u8 };

pub fn rainbow(n: usize) ![]RGB {
    var out = try allocator.alloc(RGB, n);
    for (out, 0..) |*px, i| px.* = .{ .r = @intCast(i * 255 / n), .g = 128, .b = 255 };
    return out;
}
